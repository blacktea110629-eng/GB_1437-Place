\section*{一個驚人的證明：$1+1=2$}

$$\begin{aligned}
& \text{求: } 1+1=?  \\
&= \sin \frac{\pi}{2} + \cos 0 \\
&= \sin x \bigg|_{0}^{\frac{\pi}{2}} + \cos x \bigg|_{\frac{\pi}{2}}^{0}\\
&=\int_0^\frac{\pi}{2}\cos x\,dx+\int_\frac{\pi}{2}^0-\sin x\,dx \\
&=\int_0^\frac{\pi}{2}(\sin x+\cos x)\,dx \\
&=\int_0^\frac{\pi}{2}\sqrt{2}\sin (x+\frac{\pi}{4})\,dx \\
&=-\sqrt2\cos(x+\frac{\pi}{4})\bigg|_{0}^{\frac{\pi}{2}}\\
&=-\sqrt2\left(\cos\left(\frac{\pi}{2}+\frac{\pi}{4}\right)-\cos\frac{\pi}{4}\right)\\
&=-\sqrt2\left(-\sin\frac{\pi}{4}-\cos\frac{\pi}{4}\right)\\
&=\sqrt2\times\sqrt2\sin\left(\frac{\pi}{4}+\frac{\pi}{4}\right)\\
&=2\sin\left(\frac{\pi}{4}(1+1)\right)\\
&\text{Let } 1+1=x\\
&x=2\sin\left(\frac{\pi}{4}x\right)\\
&\text{圖像法解方程(見第二頁)}\\
\end{aligned}$$
